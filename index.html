/** * SK일렉링크 채용 대시보드 API * 제작: Gemini */ function doGet() { try { const data = getAnalysisData(); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } catch (e) { // 에러 발생 시 브라우저에서 확인할 수 있도록 메시지 반환 return ContentService.createTextOutput(JSON.stringify({ "error": e.toString(), "debug": "시트 이름을 다시 확인하거나 ID 연결 상태를 확인하세요." })).setMimeType(ContentService.MimeType.JSON); } } function getAnalysisData() { // [필수 수정] 본인의 스프레드시트 URL에서 ID를 복사해 넣어주세요. // 예: https://docs.google.com/spreadsheets/d/123456789.../edit 이라면 "123456789..." 가 ID입니다. const SPREADSHEET_ID = 'https://docs.google.com/spreadsheets/d/1MG333XE24EUCGHOdVCk4ptWLp4uoGFNpEDIj-gpcqjg/edit?gid=867729802#gid=867729802'; const ss = SpreadsheetApp.openById(SPREADSHEET_ID); // 시트 이름 설정 (공백 유무를 여기서 조절할 수 있습니다) const POS_SHEET_NAME = '포지션현황'; const CAND_SHEET_NAME = '후보자현황'; const posSheet = ss.getSheetByName(POS_SHEET_NAME); const candSheet = ss.getSheetByName(CAND_SHEET_NAME); // 시트 존재 여부 체크 if (!posSheet) throw new Error("'" + POS_SHEET_NAME + "' 시트를 찾을 수 없습니다."); if (!candSheet) throw new Error("'" + CAND_SHEET_NAME + "' 시트를 찾을 수 없습니다."); // 데이터 가져오기 const posData = posSheet.getDataRange().getValues(); const candData = candSheet.getDataRange().getValues(); return { status: "success", positions: posData, candidates: candData, timestamp: new Date().getTime() }; }
